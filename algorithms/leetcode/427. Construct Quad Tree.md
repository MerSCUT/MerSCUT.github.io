# 427. 矩阵压缩四叉树

这是第一版自顶向下的代码, 但是**这种做法会在不同的递归层次中产生大量的重复扫描**.

```cpp
class Solution {
public:

    Node* construct(vector<vector<int>>& grid) {
        int len = grid.size();
        return dfs(grid, 0, 0, len);
    }
    
    Node* dfs(vector<vector<int>>& grid, int tlx, int tly, int len){
        Node* root;
        // 边界
        if (len == 1){
            return new Node(grid[tlx][tly], true);
        }
        int com = grid[tlx][tly];
        int tlxx = tlx + len/2;
        int tlyy = tly + len/2;
        int brx = tlx + len, bry = tly + len;
        for(int i = tlx; i < brx; i++)
        for(int j = tly; j < bry; j++){
            if (grid[i][j] != com){
                // 递归
                root = new Node(false, false);      // 中间结点
                root->topLeft = dfs(grid, tlx, tly, len/2);
                root->topRight = dfs(grid, tlx, tlyy, len/2);
                root->bottomLeft = dfs(grid, tlxx, tly, len/2);
                root->bottomRight = dfs(grid, tlxx, tlyy, len/2);
                return root;
            }
        }
        
        root = new Node(com, true);
        return root;
    }
};
```

为了充分发挥四叉树中标记相同值的优势, 无脑拆分到最细, 然后合并, 也就是**自底向上**可以避免重复扫描.

但这种做法最终在 Leetcode 中的时空表现一般.

分析下来可能与几方面有关

- Leetcode 的测试用例的确具有大量的重复块数字.
- 该算法实际用于**图片/材质压缩**, 而其中也会出现大量重复色块, 使得自顶向下可以更早触发剪枝.
- 重复的`new`和`delete`十分消耗时间和内存.

```cpp
class Solution {
public:

    Node* construct(vector<vector<int>>& grid) {
        int len = grid.size();
        // 自底向上应该是操作指针的.
        return build(grid, 0, 0, len);
    }

    Node* build(vector<vector<int>>& grid, int x, int y, int len){
        Node* root;
        if (len == 1){
            // 处理 root
            root = new Node(grid[x][y], true);
            return root;
        }
        // 其他情况直接无脑建子树
        root = new Node(false, false);
        root->topLeft = build(grid, x, y, len/2);
        root->topRight = build(grid, x, y + len/2, len/2);
        root->bottomLeft = build(grid, x + len/2, y, len/2);
        root->bottomRight = build(grid, x + len/2, y + len/2, len/2);
        int com = root->topLeft->val;
        if (root->topLeft->isLeaf && root->topRight->isLeaf && 
        root->bottomLeft->isLeaf && root->bottomRight->isLeaf &&
        root->topRight->val == com && root->bottomLeft->val == com &&
        root->bottomRight->val == com){
            // 合并
            root->val = com;
            root->isLeaf = true;
            delete root->topLeft;
            delete root->topRight;
            delete root->bottomLeft;
            delete root->bottomRight;
            root->topLeft = nullptr;
            root->topRight = nullptr;
            root->bottomLeft = nullptr;
            root->bottomRight = nullptr;
        }
        return root;
    }
    
    
};
```

 另一种避免重复扫描的思路是利用**二维前缀和**.

-  预处理原数组, 求二维前缀和
-  能够以 $O(1)$ 时间查询任何子矩形的和. 而这题刚好需要判断的是**全1和全0**, 它们有特殊的前缀和值.

最终代码为 :

```cpp
class Solution {
public:

    Node* construct(vector<vector<int>>& grid) {
        int len = grid.size();
        vector<vector<int>> pre(len + 1, vector<int>(len + 1, 0));
        // pre[i][j] 是 grid[i-1][j-1] 到 grid[0][0] 的和
        // pre[i][j] = pre[i-1][j] + pre[i][j-1] - pre[i][j] + grid[i-1][j-1]
        for(int i = 1; i < len + 1; i++){
            for(int j = 1; j < len + 1; j++){   
                pre[i][j] = pre[i-1][j] + pre[i][j-1] - pre[i-1][j-1] + grid[i-1][j-1];
            }
        }
        return dfs(pre, 0, 0, len);
    }
    
    Node* dfs(vector<vector<int>>& pre, int tlx, int tly, int len){
        Node* root;
        int tlxx = tlx + len/2;
        int tlyy = tly + len/2;
        int brx = tlx + len, bry = tly + len;
        // 二重循环遍历改为 前缀和直接O(1)查 grid[tlx][tly] 到 gird[brx-1][bry-1];
        int sum = pre[brx][bry] + pre[tlx][tly] - pre[brx][tly] - pre[tlx][bry];
        if (sum != len * len && sum != 0)){
            // 递归
            root = new Node(false, false);      // 中间结点
            root->topLeft = dfs(pre, tlx, tly, len/2);
            root->topRight = dfs(pre, tlx, tlyy, len/2);
            root->bottomLeft = dfs(pre, tlxx, tly, len/2);
            root->bottomRight = dfs(pre, tlxx, tlyy, len/2);
            return root;
        }
        root = new Node(sum, true);
        return root;
    }
};
```
